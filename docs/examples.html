End-to-end workflows showing how DRUID handles real data problems.

Quickstart

The minimal DRUID workflow — six function calls from raw data to model recommendations.

quickstart.py
import druid

# 1. Configure your AI provider (pick one)
druid.configure(provider="openai")
# druid.configure(provider="anthropic")
# druid.configure(provider="google")

# 2. Load your data (auto-detects format)
ds = druid.load("your_data.csv", target="label_column")

# 3. AI inspects the schema
druid.inspect(ds)

# 4. Guided EDA — plots + AI commentary
druid.explore(ds, save_dir="./eda_output")

# 5. One-line preprocessing
ds = druid.prepare(ds)

# 6. Train and compare models
results = druid.experiment(ds)

# 7. AI recommends next steps
druid.recommend(ds, results)

# 8. Export as a reproducible script
from druid.utils import export_script
export_script(ds, "reproducible_pipeline.py")
What happens under the hood

inspect() builds a statistical profile and sends it to the LLM. The AI flags quality issues, type mismatches, and potential label leakage. explore() generates missing-value plots, distributions, correlations, and target analysis. prepare() chains cleaning, imputation, outlier treatment, and encoding with sensible defaults. experiment() trains 7+ algorithms and returns a ranked leaderboard.

Fraud Detection

A production-grade fraud detection workflow with custom preprocessing and detailed model evaluation.

fraud_detection.py
import druid
from druid.preprocessing import (
    auto_clean, featurize_datetime,
    treat_outliers, encode_categoricals,
    bin_numeric, split_data,
)
from druid.modelling import classification_summary, feature_importance

# Configure AI (Claude in this case)
druid.configure(provider="anthropic")

# Load from BigQuery
ds = druid.load(
    "myproject.training_data.fraud_features",
    target="fraud_label",
    name="fraud_training",
)

# AI schema inspection
inspection = druid.inspect(ds)

# Explore — ask a specific question
eda = druid.explore(
    ds,
    question="What patterns distinguish fraudulent transactions?",
)

Instead of the one-line druid.prepare(), this example shows granular preprocessing for more control:

Custom preprocessing
# Step-by-step preprocessing
ds = auto_clean(ds)

# Extract temporal features from date columns
ds = featurize_datetime(ds, columns=["txn_date", "date_joined"])

# Custom age bins (domain-specific)
ds = bin_numeric(
    ds,
    column="age",
    bins=[0, 2, 12, 25, 45, 65, 120],
    labels=["baby", "child", "young_adult", "mid_age", "senior", "elderly"],
)

# Outlier treatment and encoding
ds = treat_outliers(ds)
ds = encode_categoricals(ds)

Train models and evaluate the best one in detail:

Model evaluation
# Train and compare
results = druid.experiment(ds, task_type="classification")

# Detailed evaluation of the top model
X_train, X_test, y_train, y_test = split_data(ds)
best = results["best_model"]
summary = classification_summary(best, X_test.values, y_test.values)
print(summary["report"])

# Feature importance
fi_fig = feature_importance(best, list(X_train.columns))

# AI recommends what to tune
druid.recommend(ds, results)

Using DRUID Without AI

DRUID works fully without an API key. Skip druid.configure() and all the automated tools still run — profiling, visualisations, preprocessing, and model experiments.

no_ai.py
import druid

# No configure() call — everything still works
ds = druid.load("customers.parquet", target="churn")

# Prints statistical profile (no AI commentary)
druid.inspect(ds)

# Generates all plots (no AI guidance text)
druid.explore(ds, save_dir="./plots")

# Preprocessing — works exactly the same
ds = druid.prepare(ds)

# Model experiments — full leaderboard
results = druid.experiment(ds)

# recommend() will remind you to add an API key
druid.recommend(ds, results)

Custom Pipeline with Session Export

Build a preprocessing pipeline step-by-step, then export the entire session as a standalone script.

custom_pipeline.py
import druid
from druid.preprocessing import (
    auto_clean, featurize_datetime,
    treat_outliers, encode_categoricals,
)
from druid.utils import export_script, save_dataset

druid.configure(provider="google")

ds = druid.load("sales.xlsx", target="revenue_bucket")

# Each step is recorded in the session
ds = auto_clean(ds)
ds = featurize_datetime(ds, columns=["order_date"])
ds = treat_outliers(ds, method="gaussian", fold=3.0)
ds = encode_categoricals(ds)

# Run experiments
results = druid.experiment(ds)

# Save a checkpoint
save_dataset(ds, "checkpoint.pkl")

# Export the full session as a Python script
export_script(ds, "sales_pipeline.py")
# → Produces a standalone .py file with every step
